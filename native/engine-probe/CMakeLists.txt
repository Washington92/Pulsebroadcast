cmake_minimum_required(VERSION 3.24)
project(wmix_engine_probe LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Windows x64 required")
endif()
if(NOT MSVC)
  message(FATAL_ERROR "MSVC required")
endif()
set(OBS_HEADERS "" CACHE PATH "Path to matching OBS source libobs directory")
set(OBS_BUILD_HEADERS "" CACHE PATH "Path containing generated obs-config.h (if needed)")
if(NOT EXISTS "${OBS_HEADERS}/obs.h")
  message(FATAL_ERROR "Set -DOBS_HEADERS=C:/src/obs-studio/libobs (matching installed OBS version)")
endif()
add_executable(engine-probe main.cpp)
target_compile_features(engine-probe PRIVATE cxx_std_17)
target_compile_definitions(engine-probe PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_include_directories(engine-probe PRIVATE "${OBS_HEADERS}" "${OBS_BUILD_HEADERS}")
target_link_libraries(engine-probe PRIVATE kernel32)
if(MSVC)
  target_compile_options(engine-probe PRIVATE /W4 /utf-8)
endif()
